async function enrichBatch(batch, fields){ // batchSize recommandé = 1 const item = batch[0]; const company = item.company; const geo = detectGeo(item.ctx, item.row) || 'Switzerland'; const prompt = ` You are a precise Swiss business contact researcher. Your task: Find official business contact information for ONE company in Switzerland. COMPANY: "${company}" KNOWN CONTEXT: ${JSON.stringify(item.ctx || {}, null, 2)} GEOGRAPHIC HINT: ${geo} FIELDS TO FIND: ${fields.join(', ')} STRICT RULES: - Search ONLY for this company - Prioritize Switzerland - Prefer official sources first: 1. official website 2. local.ch 3. search.ch 4. moneyhouse.ch 5. Google business-like listings / trusted directories - Never invent data - If unsure, return null - If multiple companies have similar names, ONLY keep the one matching the Swiss context - website = domain only (example.ch) - email = lowercase - phone = best official business number - address = full postal address if found - Add a confidence level for each field: high / medium / low Return ONLY valid JSON: { "website": null, "website_confidence": "low", "email": null, "email_confidence": "low", "phone": null, "phone_confidence": "low", "address": null, "address_confidence": "low" } `; const body = { model: MODEL, max_tokens: 1200, tools: [{ type: 'web_search_20250305', name: 'web_search' }], messages: [{ role: 'user', content: prompt }] }; const d = await callAPI(body); const raw = (d.content || []) .filter(b => b.type === 'text') .map(b => b.text) .join(''); try { const start = raw.indexOf('{'); const end = raw.lastIndexOf('}'); if(start >= 0 && end > start){ const obj = JSON.parse(raw.slice(start, end + 1)); return [{ ok: true, data: sanitize(obj || {}), confidence: { website: obj.website_confidence || 'low', email: obj.email_confidence || 'low', phone: obj.phone_confidence || 'low', address: obj.address_confidence || 'low' }, geo }]; } } catch (e) {} return [{ ok: false, data: null, confidence: {}, geo }]; }
